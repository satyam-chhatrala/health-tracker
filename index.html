// Add your deployed Cloud Run URL here later. Use localhost for testing. const API_URL = 'http://localhost:8080/api'; let storage = {}; // Now purely in-memory // 1. Update init() to fetch from DB on load async function init() { document.getElementById('date-picker').value = '2026-06-06'; document.getElementById('date-picker').onchange = (e) => { loadData(e.target.value); if(activeTab === 'analytics') generateCharts(); }; // Fetch DB data before loading the UI try { const response = await fetch(`${API_URL}/logs`); if (response.ok) { storage = await response.json(); } } catch (err) { console.error("Failed to fetch from DB, falling back to empty state", err); } loadData('2026-06-06'); } // 2. Remove localStorage.setItem from autoSave function autoSave() { const date = document.getElementById('date-picker').value; const split = document.getElementById('split-selector').value; const existingData = storage[date] || { sets: {}, cardio: {} }; const selectedFoods = FOOD_DB.filter(f => document.getElementById(`chk-${f.id}`)?.checked).map(f => f.id); const water = document.getElementById('water-in').value; const steps = document.getElementById('steps-in').value; const junkKcal = document.getElementById('junk-kcal').value; const junkProtein = document.getElementById('junk-protein').value; const sets = {}; if (split !== 'Cardio') { WORKOUT_DB[split].forEach(ex => { sets[`${ex}-s1`] = document.getElementById(`${ex}-s1`)?.value || ''; sets[`${ex}-s2`] = document.getElementById(`${ex}-s2`)?.value || ''; sets[`${ex}-s3`] = document.getElementById(`${ex}-s3`)?.value || ''; }); } const updatedSets = { ...existingData.sets, ...sets }; const cardio = { incline: document.getElementById('cardio-incline') ? document.getElementById('cardio-incline').value : existingData.cardio?.incline || '', speed: document.getElementById('cardio-speed') ? document.getElementById('cardio-speed').value : existingData.cardio?.speed || '', minutes: document.getElementById('cardio-minutes') ? document.getElementById('cardio-minutes').value : existingData.cardio?.minutes || '', kcal: document.getElementById('cardio-kcal') ? document.getElementById('cardio-kcal').value : existingData.cardio?.kcal || '' }; // Update in-memory storage ONLY storage[date] = { split, foods: selectedFoods, sets: updatedSets, water, steps, junkKcal, junkProtein, cardio }; updateMetrics(); } // 3. Update synchronizeData() to push to DB async function synchronizeData() { autoSave(); // Ensure memory is up to date const date = document.getElementById('date-picker').value; const btn = document.querySelector('button[onclick="synchronizeData()"]'); btn.innerText = 'SYNCING TO CLOUD...'; try { const response = await fetch(`${API_URL}/log`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ date: date, data: storage[date] }) }); if (response.ok) { btn.innerText = 'DATA SYNCHRONIZED ✓'; btn.classList.add('bg-white', 'text-black'); } else { throw new Error('Sync failed'); } } catch (error) { console.error(error); btn.innerText = 'SYNC FAILED ❌'; btn.classList.add('bg-red-500', 'text-white'); } setTimeout(() => { btn.innerText = 'SYNCHRONIZE DATA'; btn.classList.remove('bg-white', 'bg-red-500', 'text-white'); }, 2000); }
